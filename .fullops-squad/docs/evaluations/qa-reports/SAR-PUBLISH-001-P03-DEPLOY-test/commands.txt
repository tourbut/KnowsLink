##### deploy.txt
set -o errexit -o nounset -o pipefail
D=/home/shin/deploy/knowslink; S=/home/shin/deploy/knowslink-state; W=$S/p03deploy-20261022
NEW=4fdd149b27d3ff251a73128120dce9f8bb136620; FIX=knowslink-p03deploy-rehearse-20261022
export KNOWSLINK_STATE_DIR=$S KNOWSLINK_UID=$(id -u) KNOWSLINK_GID=$(id -g)
cd "$D"
test "$(git rev-parse HEAD)" = "$NEW"
test -z "$(git status --porcelain)"
test "$(docker inspect -f '{{.State.Health.Status}}' knowslink-relay-1)" = healthy
STOPPED=0; UPDONE=0
trap 'rc=$?; if [ $rc -ne 0 ] && [ "$STOPPED" = 1 ] && [ "$UPDONE" = 0 ]; then echo "TRAP: pre-up failure rc=$rc; restarting old relay container"; docker start knowslink-relay-1 || true; fi' EXIT
umask 077
bash "$W/observe.txt" >"$W/before.json"
STOPPED=1
echo "STOP_BEGIN=$(date -u +%FT%TZ)"
docker compose -p knowslink -f compose.yaml -f deploy/knowslink/compose.ops.yaml --env-file "$S/.env" stop relay
E1=$(docker exec knowslink-postgres-1 psql -X -U knowslink -Atc 'select epoch from relay_state')
sleep 2
E2=$(docker exec knowslink-postgres-1 psql -X -U knowslink -Atc 'select epoch from relay_state')
test "$E1" = "$E2"; echo "FROZEN_EPOCH=$E1"
OTHER=$(docker exec knowslink-postgres-1 psql -X -U knowslink -Atc "select count(*) from pg_stat_activity where datname='knowslink' and backend_type='client backend' and pid<>pg_backend_pid()")
test "$OTHER" = 0; echo "OTHER_DB_CLIENTS=$OTHER"
FROZEN=$(docker exec knowslink-postgres-1 psql -X -U knowslink -Atc 'select md5(data::text) from relay_state')
printf '%s\n' "$FROZEN" >"$W/frozen-data.md5"
bash "$W/observe.txt" >"$W/frozen.json"
OUT=$S/backups/7c6ef373-pre-p03deploy-$(date -u +%Y%m%dT%H%M%SZ).dump
(set -o noclobber; docker exec knowslink-postgres-1 pg_dump -U knowslink -Fc knowslink >"$OUT")
printf '%s\n' "$OUT" >"$W/backup.path"
docker exec -i knowslink-postgres-1 pg_restore --list <"$OUT" >"$W/backup.list"
stat -c 'backup mode=%a bytes=%s' "$OUT"
sha256sum "$OUT"
printf 'TOC_LINES='; wc -l <"$W/backup.list"
IMG=postgres:17-alpine@sha256:b0f9560a2de083e2cc7382e75f808c7381a32852a7ec49117deedb300e552b24
docker run -i -e EXPECTED_MD5="$FROZEN" --network none --name "$FIX" --user postgres --entrypoint sh "$IMG" -c '
set -eu
initdb -D /tmp/pgd -A trust >/dev/null
pg_ctl -D /tmp/pgd -o "-c listen_addresses= -c unix_socket_directories=/tmp" -w -l /tmp/pg.log start >/dev/null
export PGHOST=/tmp
createdb rehearse
pg_restore -d rehearse --no-owner --no-privileges --exit-on-error
ACTUAL_MD5=$(psql -d rehearse -Atc "select md5(data::text) from relay_state")
test "$ACTUAL_MD5" = "$EXPECTED_MD5"; echo "RESTORED_FULL_STATE_DIGEST_MATCH=true"
psql -d rehearse -v ON_ERROR_STOP=1 -Atc "select '\''restored goose='\''||(select max(version_id) from goose_db_version)||'\'' guard='\''||(select count(*) from pg_trigger where tgname='\''human_writer_guard'\'')||'\'' rows='\''||count(*)||'\'' epoch='\''||max(epoch)||'\'' HumanModel='\''||max(data->>'\''HumanModel'\'') from relay_state"
if psql -d rehearse -v ON_ERROR_STOP=1 -Atc "update relay_state set data=data-'\''HumanModel'\''" >/tmp/guard.log 2>&1; then
 echo "FAIL downgrade accepted"; exit 42
else
 grep -q "human authorization writer downgrade refused" /tmp/guard.log
 echo "restored writer downgrade refused"
fi
pg_ctl -D /tmp/pgd -m fast stop >/dev/null
' <"$OUT"
echo "REHEARSAL_EXIT=0 retained=$FIX"
UPDONE=1
docker compose -p knowslink -f compose.yaml -f deploy/knowslink/compose.ops.yaml --env-file "$S/.env" up -d --no-build --wait relay
docker compose -p knowslink -f compose.yaml -f deploy/knowslink/compose.ops.yaml --env-file "$S/.env" ps -a
echo "UP_DONE=$(date -u +%FT%TZ)"
python3 "$D/deploy/knowslink/verify.py" local
printf '%s 7c6ef3734a38a8f273714aa2915dc2f14f83a7e4 -> 4fdd149b27d3ff251a73128120dce9f8bb136620 release manual verified bundle+backup+retained-isolation+rollback-tag\n' "$(date -u +%FT%TZ)" >>"$S/deploy-history.log"
bash "$W/observe.txt" >"$W/after.json"
docker inspect -f '{{.Name}} {{.Image}} {{.State.Health.Status}} restarts={{.RestartCount}}' knowslink-relay-1
printf 'MIGRATE_EXIT='; docker inspect -f '{{.State.ExitCode}}' knowslink-migrate-1
docker logs knowslink-migrate-1
echo "RELAY_LOG_LINES=$(docker logs knowslink-relay-1 2>&1 | wc -l)"
echo "RELAY_ERROR_PANIC=$(docker logs knowslink-relay-1 2>&1 | grep -c -i -E 'error|panic' || true)"
echo 'DEPLOY_EXIT=0'
