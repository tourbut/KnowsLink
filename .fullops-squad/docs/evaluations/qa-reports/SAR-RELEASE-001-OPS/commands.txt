"""Read-only public smoke for release001: no credentials; bodies, cookies and redirect URLs are not retained."""

def call(host, method, path, body=None):
    c = http.client.HTTPSConnection(host, timeout=20)
    c.request(method, path, body=body, headers={'Content-Type': 'application/json'} if body else {})
    r = c.getresponse(); data = r.read(); h = r.getheader; hdr = r.getheaders(); c.close()
    return r, data, h, hdr
for spec in old['probes']:
    body = json.dumps({'jsonrpc': '2.0', 'id': 1, 'method': 'initialize', 'params': {'protocolVersion': '2025-03-26', 'capabilities': {}, 'clientInfo': {'name': 'unauthenticated-smoke', 'version': '1'}}}) if spec['method'] == 'POST' else None
    r, _, h, _ = call(spec['host'], spec['method'], spec['path'], body)
    v = {k: spec[k] for k in ['host', 'method', 'path', 'expected']}
    v.update(status=r.status, content_type=h('Content-Type'), cache_control=h('Cache-Control'), no_referrer=h('Referrer-Policy') == 'no-referrer', csp=bool(h('Content-Security-Policy')), nosniff=h('X-Content-Type-Options') == 'nosniff', location_expired=h('Location') == '/?n=expired')
    v['pass'] = v['status'] == v['expected']
    if spec['host'] == 'link.knowslog.com' and spec['path'] in ['/', '/home', '/home/friends', '/home/agents']:
        v['pass'] = v['pass'] and v['cache_control'] == 'no-store' and v['no_referrer'] and v['csp'] and v['nosniff']

import re
scripts = [(re.search(r'src="([^"]*)"', attrs) or [None, ''])[1] for attrs in re.findall(r'<script([^>]*)>', t)]

===== preservation.py =====
"""Compare baseline/before/frozen/after/final observations; writes redacted preservation.json (counts, digests, booleans only)."""
import json, pathlib
root = pathlib.Path(__file__).parent
STAGES = ['baseline', 'before', 'frozen', 'after', 'final']
o = {n: json.loads((root / (n + '.json')).read_text()) for n in STAGES}
A = []
def check(name, ok, detail=None):
    A.append({'assertion': name, 'pass': bool(ok), **({'detail': detail} if detail is not None else {})})
base, b, f, a, z = (o[n] for n in STAGES)
VOL = {'Rates'}
for k in b['digests']:
    if k in VOL:
        continue
    check(f'state map {k} digest equal at all 5 stages', len({o[n]['digests'][k] for n in o}) == 1)
check('state map counts equal at all stages except volatile Rates', all({k: v for k, v in o[n]['counts'].items() if k not in VOL} == {k: v for k, v in b['counts'].items() if k not in VOL} for n in o))
check('Remote map digests/counts/inbox/subscription metadata equal at all stages', all(o[n][k] == b[k] for n in o for k in ['remote_digests', 'remote_counts', 'remote_inbox_count', 'subscription_metadata']))
check('schema sha256 equal at all stages', len({o[n]['schema_sha256'] for n in o}) == 1)
check('HumanModel=1 and goose2/guard1/rows1 at all stages', all(o[n]['human_model'] == 1 and json.loads(o[n]['goose_trigger']) == {'goose': 2, 'guard': 1, 'rows': 1} for n in o))
check('frozen equals before for every state digest incl. Rates (writer stopped)', b['digests'] == f['digests'] and b['remote_digests'] == f['remote_digests'])
check('Rates is volatile (informational, not a permission change)', True, {'Rates_counts': [o[n]['counts']['Rates'] for n in STAGES], 'digest_equal_before_frozen': b['digests']['Rates'] == f['digests']['Rates'], 'digest_equal_frozen_after': f['digests']['Rates'] == a['digests']['Rates']})
fe = {n: o[n]['files'] for n in o}
check('.env/tunnel.uuid/access.aud hash+mode equal at all stages', all({k: v for k, v in fe[n].items() if k != 'tunnel/config.yml'} == {k: v for k, v in fe['before'].items() if k != 'tunnel/config.yml'} for n in o))
check('tunnel/config.yml unchanged through the relay switch', fe['baseline']['tunnel/config.yml'] == fe['before']['tunnel/config.yml'] == fe['after']['tunnel/config.yml'])
check('tunnel/config.yml changed only by the approved ingress edit, mode kept', fe['final']['tunnel/config.yml']['sha256'] != fe['after']['tunnel/config.yml']['sha256'] and fe['final']['tunnel/config.yml']['mode'] == fe['after']['tunnel/config.yml']['mode'])
check('tunnel credential json hash+mode equal at all stages', all(o[n]['private_tunnel_files'] == b['private_tunnel_files'] for n in o))
check('state and backups directory modes 0700 at all stages', all(o[n]['state_mode'] == '0o700' and o[n]['backups_mode'] == '0o700' for n in o))
bi = b['backup_inventory']
new = [k for k in a['backup_inventory'] if k not in bi]
check('existing 16 backups preserved (hash/mode/bytes); exactly one new backup', len(bi) == 16 and len(new) == 1 and all(a['backup_inventory'][k] == v for k, v in bi.items()) and all(z['backup_inventory'].get(k) == v for k, v in a['backup_inventory'].items()))
check('new backup is mode 0600', a['backup_inventory'][new[0]]['mode'] == '0o600', {'bytes': a['backup_inventory'][new[0]]['bytes'], 'sha256': a['backup_inventory'][new[0]]['sha256']})
cmap = lambda x: {c['name']: c for c in x['containers']}
cb, ca, cz = cmap(b), cmap(a), cmap(z)
key = ('image', 'id', 'running', 'health', 'started', 'restarts')
sig = lambda c: {k: c[k] for k in key}
changed_after = sorted(n for n in cb if sig(cb[n]) != sig(ca[n]))
check('only relay and migrate containers changed across the switch', changed_after == ['/knowslink-migrate-1', '/knowslink-relay-1'], changed_after)
check('Postgres container unchanged across switch and final', all(sig(c['/knowslink-postgres-1']) == sig(cb['/knowslink-postgres-1']) for c in (ca, cz)))
changed_final = sorted(n for n in cb if sig(cb[n]) != sig(cz[n]))
check('final: only relay, migrate and (approved restart) cloudflared changed', changed_final == ['/knowslink-cloudflared-1', '/knowslink-migrate-1', '/knowslink-relay-1'], changed_final)
check('final: the other pre-existing containers (shared services included) unchanged', len(cb) == 50 and all(sig(cb[n]) == sig(cz[n]) for n in cb if n not in changed_final), {'pre_existing': len(cb), 'unchanged': len(cb) - len(changed_final)})
check('only new container since baseline is the retained isolated restore fixture', sorted(n for n in cz if n not in cb) == ['/knowslink-release001-rehearse-20261017'])
check('relay image changed and relay healthy with restarts 0', cb['/knowslink-relay-1']['image'] != cz['/knowslink-relay-1']['image'] and cz['/knowslink-relay-1']['health'] == 'healthy' and cz['/knowslink-relay-1']['restarts'] == 0)
check('shared orca/s8/mcp statuses and host cloudflared PIDs equal at all stages', all(o[n]['shared'] == b['shared'] and o[n]['host_cloudflared_pids'] == b['host_cloudflared_pids'] for n in o), {'host_cloudflared_pids': b['host_cloudflared_pids']})
check('listeners 127.0.0.1:8080 and 5432 restored; frozen had only 5432', b['listeners'] == a['listeners'] == z['listeners'] and f['listeners'] == ['127.0.0.1:5432'])
check('no other active DB writer at frozen/after/final', all(json.loads(o[n]['db_connections'])['active_other_writers'] == 0 for n in ['frozen', 'after', 'final']))
check('deployed HEAD 1dd6800 clean at before/frozen/after/final; baseline at 8a93e38', all(o[n]['head'] == '1dd6800d5d4238e2060f99e98eaab437ea06682c' and not o[n]['dirty'] for n in ['before', 'frozen', 'after', 'final']) and base['head'] == '8a93e38e190f1b9e7b6bb183e578e410f5dd97c6')
epochs = [o[n]['epoch'] for n in STAGES]
check('epoch never decreased; stable while writer stopped', epochs == sorted(epochs), epochs)
out = {'utc_stages': {n: o[n]['utc'] for n in STAGES}, 'passed': sum(x['pass'] for x in A), 'total': len(A), 'assertions': A}
(root / 'preservation.json').write_text(json.dumps(out, indent=2, ensure_ascii=False) + '\n', encoding='utf-8')
print(out['passed'], '/', out['total'])
for x in A:
    if not x['pass']:
        print('FAIL', x['assertion'], x.get('detail'))
assert out['passed'] == out['total']
