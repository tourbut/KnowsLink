package relay

import (
	"fmt"
	"net"
	"net/http"
	"net/http/httptest"
	"strings"
	"testing"
	"time"
)

// Review repro: unauthenticated slow bodies hold process admission slots before any rate or auth check.
func TestReviewSlowBodyHoldsSlots(t *testing.T) {
	srv := httptest.NewServer((&Service{}).Handler())
	defer srv.Close()
	addr := strings.TrimPrefix(srv.URL, "http://")
	hold := func(path string, n int) []net.Conn {
		var cs []net.Conn
		for i := 0; i < n; i++ {
			c, err := net.Dial("tcp", addr)
			if err != nil {
				t.Fatal(err)
			}
			fmt.Fprintf(c, "POST %s HTTP/1.1\r\nHost: x\r\nContent-Type: application/json\r\nContent-Length: 100\r\n\r\n{", path)
			cs = append(cs, c)
		}
		time.Sleep(200 * time.Millisecond)
		return cs
	}
	probe := func(path string) (int, string) {
		resp, err := http.Post(srv.URL+path, "application/json", strings.NewReader(`{}`))
		if err != nil {
			t.Fatal(err)
		}
		defer resp.Body.Close()
		b := make([]byte, 200)
		n, _ := resp.Body.Read(b)
		return resp.StatusCode, string(b[:n])
	}
	for _, c := range hold("/v1/ack", 4) {
		defer c.Close()
	}
	code, body := probe("/v1/text/ack")
	t.Logf("cleanup probe after 4 anonymous slow bodies: %d %s", code, body)
	if code != 429 || !strings.Contains(body, "capacity") {
		t.Fatalf("expected cleanup capacity refusal, got %d %s", code, body)
	}
	for _, c := range hold("/auth/start", 16) {
		defer c.Close()
	}
	code, body = probe("/v1/text/send")
	t.Logf("new-work probe after 16 anonymous slow bodies: %d %s", code, body)
	if code != 429 || !strings.Contains(body, "capacity") {
		t.Fatalf("expected new-work capacity refusal, got %d %s", code, body)
	}
}
