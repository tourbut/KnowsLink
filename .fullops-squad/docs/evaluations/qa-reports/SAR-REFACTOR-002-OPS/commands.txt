"""Read-only public protection smoke; no credentials or response bodies are retained."""
import datetime,http.client,json,pathlib
root=pathlib.Path(__file__).parent
old=json.loads((root.parent/'refactor001-ops/public-smoke.json').read_text())
report={'utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'probes':[]}
for spec in old['probes']:
    c=http.client.HTTPSConnection(spec['host'],timeout=20)
    body=json.dumps({'jsonrpc':'2.0','id':1,'method':'initialize','params':{'protocolVersion':'2025-03-26','capabilities':{},'clientInfo':{'name':'unauthenticated-smoke','version':'1'}}}) if spec['method']=='POST' else None
    c.request(spec['method'],spec['path'],body=body,headers={'Content-Type':'application/json'} if body else {})
    r=c.getresponse()
    v={k:spec[k] for k in ['host','method','path','expected']}
    v.update(status=r.status,content_type=r.getheader('Content-Type'),cache_control=r.getheader('Cache-Control'),no_referrer=r.getheader('Referrer-Policy')=='no-referrer',csp=bool(r.getheader('Content-Security-Policy')),nosniff=r.getheader('X-Content-Type-Options')=='nosniff',location_expired=r.getheader('Location')=='/?n=expired')
    r.read();c.close()
    v['pass']=v['status']==v['expected']
    if spec['host']=='link.knowslog.com' and spec['path'] in ['/','/home','/home/friends','/home/agents']:
        v['pass']=v['pass'] and v['cache_control']=='no-store' and v['no_referrer'] and v['csp'] and v['nosniff']
    if spec['path'].startswith('/home'): v['pass']=v['pass'] and v['location_expired']
    report['probes'].append(v)
(root/'public-smoke.json').write_text(json.dumps(report,indent=2)+'\n',encoding='utf-8')
assert all(v['pass'] for v in report['probes']), 'Public smoke mismatch; inspect redacted report'
print('Public smoke: 21/21 passed; bodies/cookies not retained')
