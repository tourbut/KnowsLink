func TestReviewFix2EqualCommitTime(t *testing.T) {
 old, newer := newState(), newState()
 s := &Service{}
 at := time.Now()
 s.remember(newer, at)
 s.remember(old, at)
 if s.live.Load().st != old { t.Fatal("equal-time observation changed") }
 t.Log("OBSERVED: equal DB timestamps allow a late older snapshot to replace a newer snapshot")
 s.remember(newer, at.Add(time.Microsecond))
 s.remember(old, at)
 if s.live.Load().st != newer { t.Fatal("strictly older snapshot replaced newer") }
}

func TestReviewFix2StaleAndNilWithNew16(t *testing.T) {
 for _, mode := range []string{"stale", "restart"} {
  t.Run(mode, func(t *testing.T) {
   pool := messagePool(t)
   f := setup(t, pool)
   original := f.s
   other := &Service{Pool: pool, SyntheticSignup:true}
   f.s, f.handler = other, other.Handler()
   f.addAgent("agent_new")
   s := original
   if mode == "restart" { s = &Service{Pool:pool} }
   h := s.boundedHTTP(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request){w.WriteHeader(200)}))
   for range 16 { s.httpNew <- struct{}{} }
   request := func() int {
    r := httptest.NewRequest("POST", "/v1/key-revoke", strings.NewReader(`{"agent":"agent_new","kid":"key1"}`))
    r.Header.Set("Authorization", "Bearer "+f.tokens["agent_new_owner"])
    w := httptest.NewRecorder(); h.ServeHTTP(w,r); return w.Code
   }
   if got:=request(); got!=429 { t.Fatalf("before refresh got %d",got) }
   ctx,cancel:=context.WithCancel(context.Background())
   done:=make(chan struct{})
   go func(){ defer close(done); s.Cleanup(ctx) }()
   deadline:=time.Now().Add(4*time.Second)
   for s.live.Load()==nil || s.live.Load().creds[hashToken(f.tokens["agent_new_owner"])].id=="" {
    if time.Now().After(deadline){cancel();<-done;t.Fatal("sweep failed to refresh snapshot")}
    time.Sleep(10*time.Millisecond)
   }
   cancel();<-done
   if got:=request();got!=200{t.Fatalf("after actual sweep got %d",got)}
   for range 16 { <-s.httpNew }
   t.Log("OBSERVED: valid cleanup before refresh=429 capacity; after actual 1s sweep=200 while local new16 remain full")
  })
 }
}

func TestReviewFix2SameOwnerAgentFlood(t *testing.T) {
 pool:=messagePool(t); f:=setup(t,pool)
 now:=time.Now()
 f.mutate(func(st *State){
  st.Rates=map[string][]time.Time{}
  st.Messages["self-lease"]=&Message{Receipt:Receipt{ID:"self-lease",From:"agent_a",To:"agent_b",Intent:"relay.request",State:"leased",Exp:now.Add(3*time.Minute),Accepted:now},Kid:"key1",Generation:st.Pairs[pairID("agent_a","agent_b")].Generation,Deliver:"agent",LeaseToken:"lease",LeaseUntil:now.Add(30*time.Second)}
 })
 h:=f.s.Handler()
 probe:=f.s.boundedHTTP(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){w.WriteHeader(200)}))
 serve:=func(h http.Handler,path,token,body string)(int,string){
  r:=httptest.NewRequest("POST",path,strings.NewReader(body));r.Header.Set("Authorization","Bearer "+token)
  w:=httptest.NewRecorder();h.ServeHTTP(w,r);return w.Code,w.Body.String()
 }
 ack:=func()(int,string){return serve(h,"/v1/ack",f.tokens["agent_b"],`{"id":"self-lease","token":"lease"}`)}
 // Deterministic: only an agent credential holds the owner's shared local cleanup unit.
 tx,err:=pool.Begin(context.Background());if err!=nil{t.Fatal(err)}
 defer func(){_ = tx.Rollback(context.Background())}()
 if _,err=tx.Exec(context.Background(),"SELECT 1 FROM relay_state WHERE singleton = true FOR UPDATE");err!=nil{t.Fatal(err)}
 done:=make(chan int,1);go func(){c,_:=ack();done<-c}()
 deadline:=time.Now().Add(time.Second)
 for len(f.s.httpClean)!=1 {if time.Now().After(deadline){t.Fatal("agent ACK did not take cleanup")};time.Sleep(time.Millisecond)}
 c,b:=serve(h,"/v1/key-revoke",f.tokens["agent_b_owner"],`{"agent":"agent_b","kid":"key1"}`)
 if c!=429 || !strings.Contains(b,"capacity"){t.Fatalf("same-owner revocation got %d %s",c,b)}
 _ = tx.Rollback(context.Background())
 if c:=<-done;c!=409{t.Fatalf("unpersisted ACK handler got %d",c)}
 t.Log("OBSERVED deterministic: agent ACK (handler invalid_lease 409) prevents own owner's key-revoke before DB, 429 capacity")
 f.mutate(func(st *State){st.Rates=map[string][]time.Time{}})
 var stop atomic.Bool;var wg sync.WaitGroup
 for range 48 {wg.Add(1);go func(){defer wg.Done();for !stop.Load(){ack()}}()}
 time.Sleep(100*time.Millisecond)
 capacity,success,rate:=0,0,0
 for range 18 {
  c,b:=serve(probe,"/v1/key-revoke",f.tokens["agent_b_owner"],`{"agent":"agent_b","kid":"key1"}`)
  if c==429&&strings.Contains(b,"capacity"){capacity++}else if c==200{success++}else{rate++}
  time.Sleep(50*time.Millisecond)
 }
 stop.Store(true);wg.Wait()
 t.Logf("OBSERVED lock-free same-owner revoke admission: capacity429=%d/18 admitted=%d other=%d",capacity,success,rate)
 if capacity==0{t.Fatal("same-owner flood did not impede owner")}
 f.mutate(func(st *State){
  t.Logf("ACK cleanup rate attempts=%d; owner's cleanup rate attempts=%d",len(st.Rates["cleanup:member:agent_b"]),len(st.Rates["cleanup:member:"+f.owners["agent_b"]]))
  st.Rates=map[string][]time.Time{}
 })
 c,b=serve(h,"/v1/key-revoke",f.tokens["agent_b_owner"],`{"agent":"agent_b","kid":"key1"}`)
 if c!=200{t.Fatalf("after flood actual revoke got %d %s",c,b)}
 t.Log("after attacker stops: actual key-revoke=200; other owners retain their cleanup slots")
}

func TestReviewFix2OrphanOtherInstanceAndSweep(t *testing.T){
 pool:=messagePool(t);f:=setup(t,pool);other:=&Service{Pool:pool}
 f.mutate(func(st *State){st.HTTP["orphan"]=admission{Clean:true,Exp:time.Now().Add(30*time.Second),Owner:f.owners["agent_a"]}})
 f.s.orphans.Store("orphan",nil)
 if _,err:=other.transaction(context.Background(),func(st *State,_ time.Time)(any,error){return nil,nil});err!=nil{t.Fatal(err)}
 if _,known:=f.s.orphans.Load("orphan");!known{t.Fatal("other instance consumed local orphan")}
 ctx,cancel:=context.WithCancel(context.Background());done:=make(chan struct{})
 go func(){defer close(done);f.s.Cleanup(ctx)}()
 deadline:=time.Now().Add(4*time.Second)
 for { _,known:=f.s.orphans.Load("orphan");if !known{break};if time.Now().After(deadline){cancel();<-done;t.Fatal("own sweep did not reclaim")};time.Sleep(10*time.Millisecond) }
 cancel();<-done
 if _,err:=other.transaction(context.Background(),func(st *State,_ time.Time)(any,error){if _,held:=st.HTTP["orphan"];held{t.Fatal("sweep left shared orphan")};return nil,nil});err!=nil{t.Fatal(err)}
 t.Log(fmt.Sprint("OBSERVED: other-instance commit retains local orphan; successful own 1s sweep removes token and shared admission"))
}
