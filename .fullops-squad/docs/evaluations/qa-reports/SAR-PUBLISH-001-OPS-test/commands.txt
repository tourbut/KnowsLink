##### public smoke scripts: public-smoke.py, public-smoke-lang.py (task-local, read-only), public-smoke-policy.py
"""Read-only public check of /privacy, /terms and neighbours for SAR-PUBLISH-001-OPS: no credentials, bodies and cookie values not retained."""
import datetime, http.client, json, pathlib, re, time
root = pathlib.Path(__file__).parent
rep = {'utc': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'policy': [], 'neighbours': [], 'calendar': [], 'landing': {}}
def call(method, path, headers=None):
    c = http.client.HTTPSConnection('link.knowslog.com', timeout=20)
    c.request(method, path, headers=headers or {})
    r = c.getresponse(); data = r.read(); hdr = r.getheaders(); c.close(); time.sleep(1.5)
    return r, data.decode('utf-8', 'replace'), hdr
for path in ['/privacy', '/terms']:
    for lang in ['ko', 'en']:
        r, t, hdr = call('GET', path, {'Cookie': '__Host-kl_lang=' + lang} if lang == 'en' else {})
        h = lambda k: next((v for kk, v in hdr if kk.lower() == k.lower()), None)
        v = {'path': path, 'lang': lang, 'status': r.status, 'content_type': h('Content-Type'), 'cache_control': h('Cache-Control'), 'content_language': h('Content-Language'),
             'set_cookie_absent': not any(k.lower() == 'set-cookie' for k, _ in hdr), 'effective_2026-10-22': '2026-10-22' in t, 'has_h1': '<h1' in t, 'no_img': '<img' not in t,
             'robots_noindex': 'noindex' in t.lower()}
        v['pass'] = r.status == 200 and (h('Content-Type') or '').startswith('text/html') and v['set_cookie_absent'] and v['effective_2026-10-22'] and v['has_h1']
        rep['policy'].append(v)
for path in ['/privacy/', '/privacy/x', '/privacyx', '/terms/', '/terms.html', '/Privacy', '/policy']:
    r, _, _ = call('GET', path)
    rep['neighbours'].append({'path': path, 'status': r.status, 'pass': r.status == 404})
for method, path in [('GET', '/home/calendar/connect'), ('POST', '/home/calendar/connect'), ('POST', '/home/calendar/disconnect'), ('GET', '/home/calendar')]:
    r, _, hdr = call(method, path)
    rep['calendar'].append({'method': method, 'path': path, 'status': r.status, 'pass': r.status == 404, 'cookies_set': [k for k, _ in hdr if k.lower() == 'set-cookie']})
# first page: policy links present, no calendar/freebusy wording, with or without login block shown
r, t, hdr = call('GET', '/')
L = rep['landing']; L['status'] = r.status
L['links'] = {p: (('href="%s"' % p) in t) for p in ['/privacy', '/terms']}
L['start_anchor'] = 'id="start"' in t
L['calendar_words_absent'] = not re.search(r'calendar|freebusy|캘린더|일정 조회', t, re.I)
L['google_form_count'] = t.count('action="/auth/google"')
L['set_cookie_absent'] = not any(k.lower() == 'set-cookie' for k, _ in hdr)
L['pass'] = r.status == 200 and all(L['links'].values()) and L['start_anchor'] and L['calendar_words_absent'] and L['set_cookie_absent']
rep['all_pass'] = all(v['pass'] for v in rep['policy'] + rep['neighbours'] + rep['calendar']) and L['pass']
(root / 'public-smoke-policy.json').write_text(json.dumps(rep, indent=2, ensure_ascii=False) + '\n', encoding='utf-8')
print([(v['path'], v['lang'], v['status']) for v in rep['policy']], [(v['path'], v['status']) for v in rep['neighbours']], [(v['method'], v['path'], v['status']) for v in rep['calendar']], L)
assert rep['all_pass'], 'policy smoke mismatch'
