// Fake orca CLI for the S6 loopback run: appends its argv as one JSON line; a "1" in <log>.fail exits 1 (mail failure).
package main

import (
	"encoding/json"
	"os"
)

func main() {
	line, _ := json.Marshal(os.Args[1:])
	f, err := os.OpenFile(os.Getenv("FAKE_ORCA_LOG"), os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0600)
	if err != nil {
		os.Exit(2)
	}
	f.Write(append(line, '\n'))
	f.Close()
	if flag, _ := os.ReadFile(os.Getenv("FAKE_ORCA_LOG") + ".fail"); string(flag) == "1" {
		os.Exit(1)
	}
}
